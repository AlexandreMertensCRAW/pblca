PBLCA — Process-Based LCA Engine.

Process-based LCA engine for mixed crop-livestock farms, ISO 14040/14044 compliant, with:

  • three distinct layers: processes (layer 1), gas flows CH4/CO2/N2O (layer 2), GWP/GWP* impacts (layer 3);
  • a model registry with a universal interface — every variant (Tier-2, Tier-3, ...) is testable at every simulation;
  • traceable parameters (bibliographic reference) with uncertainties propagated by Monte-Carlo, a given parameter keeping the same value for all its occurrences (all farms);
  • structured JSON storage (one entry per simulation);
  • explicit warnings and errors management (diagnostics).

Entry point: pblca.engine.LCAEngine.

284
285        Args:
286            farms: one farm or a list of farms (parameters are shared
287                across farms: the same draw everywhere).
288            model_selection: variant per slot (default: first registered).
289            values: set of parameter values (default: central values).
290                This mechanism guarantees that in Monte-Carlo a
291                parameter has the same value everywhere.
292            sim_id: simulation identifier (default: random).
293            record: if True, appends the entry to the JSON datastore.
294            ch4_previous_kg: CH4 emissions at t−Δt for GWP* (default:
295                equilibrium, ΔE = 0).
296
297        Returns:
298            SimulationResult (gas ledger + impacts + diagnostics).
299        """
300        if isinstance(farms, FarmContext):
301            farms = [farms]
302        farms = list(farms)
303        if not farms:
304            raise ValueError("No farm provided")
305
306        model_selection = dict(model_selection or {})
307        for slot in _PRE_MANURE_SLOTS + _POST_MANURE_SLOTS:
308            if slot not in model_selection:
309                model_selection[slot] = self.registry.get(slot).variant
310        summary = self.registry.selection_summary(model_selection)
311
312        values = values if values is not None else self.params.central_values()
313        ledger = GasLedger()
314        diagnostics = DiagLogger()
315        farm_ids: List[str] = []
316
317        # Save parcel state: run() must be idempotent (manure
318        # redistribution mutates n_organic_spread in place).
319        organic_state = [
320            (p, p.n_organic_spread) for f in farms for p in f.parcels
321        ]
322        try:
323            for farm in farms:
324                farm_ids.append(farm.farm_id)
325                ctx = ModelContext(farm, self.params, values, diagnostics)
326                n_organic = 0.0
327                # Phase 1: enteric + manure (produces the spread nitrogen).
328                for slot in _PRE_MANURE_SLOTS:
329                    spec = self.registry.get(slot, model_selection[slot])
330                    result = spec.func(ctx)
331                    source, gas = SLOT_SOURCE_GAS[slot]
332                    amount = getattr(result, f"{gas.lower()}_kg")
333                    ledger.add(
334                        gas, amount, source, farm.farm_id,
335                        spec.variant, spec.reference, detail=f"slot={slot}",
336                    )
337                    if slot == "manure_n2o":
338                        n_organic = result.fluxes.get("n_organic_available", 0.0)
339                # Redistribution of organic nitrogen to the parcels.
340                # Pasture deposits are fully handled by the manure module
341                # (EF3PRP + indirect): no transfer to the soil module,
342                # to avoid double counting.
343                if n_organic > 0:
344                    _distribute_manure_n(farm, n_organic)
345                # Phase 2: soil (N2O, carbon) + purchases + fieldwork.
346                for slot in _POST_MANURE_SLOTS:
347                    spec = self.registry.get(slot, model_selection[slot])
348                    result = spec.func(ctx)
349                    source, gas = SLOT_SOURCE_GAS[slot]
350                    amount = result.co2_kg if gas == "CO2" else result.n2o_kg
351                    if slot == "soil_carbon" and amount < 0:
352                        ledger.add_sink(
353                            amount, source, farm.farm_id,
354                            spec.variant, spec.reference, detail=f"slot={slot}",
355                        )
356                    else:
357                        ledger.add(
358                            gas, amount, source, farm.farm_id,
359                            spec.variant, spec.reference, detail=f"slot={slot}",
360                        )
361        finally:
362            # Restoration: FarmContext objects remain reusable across
363            # simulations (idempotence of run()).
364            for parcel, base in organic_state:
365                parcel.n_organic_spread = base
366
367        impacts = characterize(
368            ledger,
369            GwpStarInputs(
370                ch4_current_kg=ledger.total("CH4"),
371                ch4_previous_kg=ch4_previous_kg,
372            ),
373        )
374        sim = SimulationResult(
375            sim_id=sim_id or uuid.uuid4().hex[:12],
376            ledger=ledger,
377            impacts=impacts,
378            diagnostics=diagnostics.as_list(),
379            model_selection=model_selection,
380            farms=farm_ids,
381        )
382        if record:
383            self._record(sim, summary, values, uncertainty=None)
384        return sim

410    def run_monte_carlo(
411        self,
412        farms: Union[FarmContext, Sequence[FarmContext]],
413        n_iterations: int = 1000,
414        model_selection: Optional[Dict[str, str]] = None,
415        seed: Optional[int] = None,
416        record: bool = True,
417    ) -> Dict[str, Any]:
418        """Uncertainty propagation by Monte-Carlo.
419
420        At each iteration, ``ParameterSet.draw`` draws ONE value per
421        parameter; this value is used by every farm and every process
422        depending on that parameter (farm 1 and farm 2 share the same
423        draw — explicit requirement).
424
425        Args:
426            farms: one farm or a list of farms.
427            n_iterations: number of iterations (>0).
428            model_selection: variants per slot.
429            seed: random seed (reproducibility).
430            record: if True, records a summary entry in the datastore.
431
432        Returns:
433            statistics per indicator {mean, sd, p5, p50, p95, n} and
434            per gas, plus the central impacts and the model selection.
435        """
436        if isinstance(farms, FarmContext):
437            farms = [farms]
438        farms = list(farms)
439        rng = np.random.default_rng(seed)
440
441        # Snapshot BEFORE any run: run() mutates n_organic_spread
442        # (manure redistribution); the snapshot must capture the
443        # original state to restore the parcels at each iteration.
444        base_organic = {
445            (f.farm_id, p.key): p.n_organic_spread
446            for f in farms for p in f.parcels
447        }
448
449        def restore_parcels() -> None:
450            for f in farms:
451                for p in f.parcels:
452                    p.n_organic_spread = base_organic[(f.farm_id, p.key)]
453
454        ration_state = _ration_snapshot(farms)
455        has_measures = any(
456            dmi is not None or ge is not None for _, dmi, ge, _, _ in ration_state
457        )
458
459        # Central simulation (reference). It is recorded through the
460        # MC summary entry below (record) — not twice.
461        central_values = self.params.central_values()
462        central = self.run(
463            farms,
464            model_selection=model_selection,
465            values=central_values,
466            sim_id="central",
467            record=False,
468        )
469        summary = self.registry.selection_summary(central.model_selection)
470        restore_parcels()
471
472        impact_samples: Dict[str, List[float]] = {k: [] for k in central.impacts}
473        gas_samples: Dict[str, List[float]] = {g: [] for g in GASES}
474        failed = 0
475        for _ in range(n_iterations):
476            drawn = self.params.draw(rng)
477            restore_parcels()
478            if has_measures:
479                _perturb_measured_rations(ration_state, rng)
480            try:
481                it = self.run(
482                    farms,
483                    model_selection=central.model_selection,
484                    values=drawn,
485                    record=False,
486                )
487            except Exception:
488                failed += 1
489                continue
490            for k, v in it.impacts.items():
491                impact_samples[k].append(v)
492            for g in GASES:
493                gas_samples[g].append(it.ledger.total(g))
494        restore_parcels()
495        _set_ration_mode(ration_state, "measured")
496
497        stats = {k: _stats(v) for k, v in impact_samples.items()}
498        gas_stats = {
499            g: _stats(v) for g, v in gas_samples.items() if len(v) > 0
500        }
501        uncertainty = {
502            "method": "Monte-Carlo",
503            "n_iterations": n_iterations,
504            "seed": seed,
505            "failed_iterations": failed,
506            "impacts": stats,
507            "gas_totals_kg": gas_stats,
508        }
509        if record:
510            self._record(central, summary, central_values, uncertainty=uncertainty)
511        return {
512            "sim_id": "monte_carlo",
513            "model_selection": summary,
514            "central_impacts": central.impacts,
515            **uncertainty,
516        }

521    def run_ration_comparison(
522        self,
523        farms: Union[FarmContext, Sequence[FarmContext]],
524        n_iterations: int = 1000,
525        model_selection: Optional[Dict[str, str]] = None,
526        seed: Optional[int] = None,
527        record: bool = True,
528    ) -> Dict[str, Any]:
529        """Paired evaluation of the two ration-definition modes.
530
531        At each Monte-Carlo iteration, ONE parameter draw is performed
532        and the farms are evaluated TWICE with this same draw:
533
534        * ``ipcc_equations``: every group's measured values are
535          temporarily removed, so the IPCC energy chain (Eq. 10.3-10.16)
536          drives enteric CH4 and manure fluxes;
537        * ``measured``: the measured DMI/GE values are restored, so
538          enteric CH4 and manure fluxes rely on the farm data.
539
540        Because the two evaluations of an iteration share the same
541        parameter draw (and therefore the same EF1, Ym, B0, ...), the
542        paired difference between the two modes isolates the effect of
543        the additional ration information. The respective standard
544        deviations quantify its effect on the precision of the result.
545
546        The measured mode also propagates the quantification error of
547        the on-farm measurements: one multiplicative lognormal factor
548        per group and per iteration (median 1, sigma = the group's
549        ``ration_rel_sd``) is applied to both ``dmi_measured`` and
550        ``ge_measured``. With ``ration_rel_sd=None`` (default) the
551        measurements are treated as exact and no perturbation is
552        applied.
553
554        Args:
555            farms: one farm or a list of farms.
556            n_iterations: number of iterations (>0).
557            model_selection: variants per slot.
558            seed: random seed (reproducibility).
559            record: if True, records a summary entry in the datastore.
560
561        Returns:
562            a dictionary with, per indicator and per gas: the statistics
563            of each mode, the paired difference (measured − ipcc) and
564            the relative reduction of the standard deviation
565            (precision gain: 1 − sd_measured/sd_ipcc).
566        """
567        if isinstance(farms, FarmContext):
568            farms = [farms]
569        farms = list(farms)
570
571        # Groups without measurements are identical in both modes; a
572        # comparison is only meaningful if at least one group carries
573        # measured values.
574        ration_state = _ration_snapshot(farms)
575        if not any(dmi is not None or ge is not None for _, dmi, ge, _, _ in ration_state):
576            raise ValueError(
577                "run_ration_comparison requires at least one animal group "
578                "with dmi_measured and/or ge_measured set"
579            )
580
581        rng = np.random.default_rng(seed)
582        base_organic = {
583            (f.farm_id, p.key): p.n_organic_spread
584            for f in farms for p in f.parcels
585        }
586
587        def restore_parcels() -> None:
588            for f in farms:
589                for p in f.parcels:
590                    p.n_organic_spread = base_organic[(f.farm_id, p.key)]
591
592        modes = ("ipcc_equations", "measured")
593        # Map the context ration mode to the switch function argument.
594        mode_switch = {"ipcc_equations": "ipcc", "measured": "measured"}
595
596        central_values = self.params.central_values()
597        central = {}
598        impact_samples: Dict[str, Dict[str, List[float]]] = {
599            m: {} for m in modes
600        }
601        gas_samples: Dict[str, Dict[str, List[float]]] = {
602            m: {g: [] for g in GASES} for m in modes
603        }
604        failed = {m: 0 for m in modes}
605
606        for mode in modes:
607            _set_ration_mode(ration_state, mode_switch[mode])
608            restore_parcels()
609            run = self.run(
610                farms,
611                model_selection=model_selection,
612                values=central_values,
613                sim_id=f"central_ration_{mode}",
614                record=False,
615            )
616            central[mode] = run
617            impact_samples[mode] = {k: [] for k in run.impacts}
618
619        summary = self.registry.selection_summary(central["measured"].model_selection)
620
621        for _ in range(n_iterations):
622            drawn = self.params.draw(rng)
623            for mode in modes:
624                # The measured mode propagates the quantification error
625                # of the on-farm measurements: one lognormal factor per
626                # group and per iteration, applied to both DMI and GE.
627                # The ipcc mode ignores the measurements entirely, so
628                # the pairing on the parameter draw is preserved.
629                _set_ration_mode(ration_state, mode_switch[mode])
630                if mode == "measured":
631                    _perturb_measured_rations(ration_state, rng)
632                restore_parcels()
633                try:
634                    it = self.run(
635                        farms,
636                        model_selection=central["measured"].model_selection,
637                        values=drawn,
638                        record=False,
639                    )
640                except Exception:
641                    failed[mode] += 1
642                    continue
643                for k, v in it.impacts.items():
644                    impact_samples[mode][k].append(v)
645                for g in GASES:
646                    gas_samples[mode][g].append(it.ledger.total(g))
647
648        # Restore the original ration definition (idempotence).
649        _set_ration_mode(ration_state, "measured")
650        restore_parcels()
651
652        out: Dict[str, Any] = {
653            "sim_id": "ration_comparison",
654            "method": "paired Monte-Carlo",
655            "n_iterations": n_iterations,
656            "seed": seed,
657            "failed_iterations": failed,
658            "model_selection": summary,
659        }
660
661        indicators = list(central["measured"].impacts)
662        for indicator in indicators:
663            ipcc_s = impact_samples["ipcc_equations"][indicator]
664            meas_s = impact_samples["measured"][indicator]
665            n_pairs = min(len(ipcc_s), len(meas_s))
666            paired_diff = [
667                meas_s[i] - ipcc_s[i] for i in range(n_pairs)
668            ]
669            ipcc_stats = _stats(ipcc_s)
670            meas_stats = _stats(meas_s)
671            sd_ipcc = ipcc_stats["sd"]
672            sd_meas = meas_stats["sd"]
673            precision_gain = float("nan")
674            if sd_ipcc > 0 and not np.isnan(sd_meas):
675                precision_gain = 1.0 - sd_meas / sd_ipcc
676            out[indicator] = {
677                "ipcc_equations": ipcc_stats,
678                "measured": meas_stats,
679                "paired_difference": _stats(paired_diff),
680                "precision_gain_sd": precision_gain,
681            }
682
683        for gas in GASES:
684            ipcc_s = gas_samples["ipcc_equations"][gas]
685            meas_s = gas_samples["measured"][gas]
686            if len(ipcc_s) == 0 or len(meas_s) == 0:
687                continue
688            n_pairs = min(len(ipcc_s), len(meas_s))
689            paired_diff = [meas_s[i] - ipcc_s[i] for i in range(n_pairs)]
690            ipcc_stats = _stats(ipcc_s)
691            meas_stats = _stats(meas_s)
692            sd_ipcc = ipcc_stats["sd"]
693            sd_meas = meas_stats["sd"]
694            precision_gain = float("nan")
695            if sd_ipcc > 0 and not np.isnan(sd_meas):
696                precision_gain = 1.0 - sd_meas / sd_ipcc
697            out[f"{gas}_totals"] = {
698                "ipcc_equations": ipcc_stats,
699                "measured": meas_stats,
700                "paired_difference": _stats(paired_diff),
701                "precision_gain_sd": precision_gain,
702            }
703
704        if record:
705            self._record(
706                central["measured"],
707                summary,
708                central_values,
709                uncertainty={
710                    k: v for k, v in out.items()
711                    if k not in ("sim_id", "method")
712                },
713            )
714        return out
